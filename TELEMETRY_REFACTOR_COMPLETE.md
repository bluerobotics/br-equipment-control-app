```cpp
    // ... all 21 fields initialized

```

### 2. `telemetry_build_message(const TelemetryData* data, char* buffer, size_t buffer_size)`

Constructs complete telemetry message string:

```cpp
int telemetry_build_message(const TelemetryData* data, char* buffer, size_t buffer_size) {
    if (data == NULL || buffer == NULL || buffer_size == 0) return 0;
    
    int pos = 0;
    pos += snprintf(buffer + pos, buffer_size - pos, "%s", TELEM_PREFIX);
    
    // fillhead_state
    if (pos < buffer_size) {
        pos += snprintf(buffer + pos, buffer_size - pos, "%s:%d,", 
                       TELEM_KEY_FILLHEAD_STATE, data->fillhead_state);
    }
    
    // ... all 21 fields formatted properly
    
    return pos;
}
```

**Automatic formatting based on type:**
- `int` → `%d`
- `float` → `%.Nf` (precision from JSON)
- `bool` → `%d` (0 or 1)

### 3. `telemetry_send(const TelemetryData* data)`

Builds and transmits telemetry:

```cpp
void telemetry_send(const TelemetryData* data) {
    char buffer[512];
    int len = telemetry_build_message(data, buffer, sizeof(buffer));
    
    if (len > 0) {
        ConnectorUsb.SendLine(buffer);
    }
}
```

## Usage in Embedded Firmware

### Complete Example

```cpp
#include "telemetry.h"
#include "responses.h"

// Global telemetry structure - ALL variables in ONE place!
TelemetryData g_telemetry;

void setup() {
    // Initialize with defaults from telemetry.json
    telemetry_init(&g_telemetry);
    
    ConnectorUsb.Mode(Connector::USB_CDC);
    ConnectorUsb.PortOpen();
}

void loop() {
    // Update telemetry fields as application runs
    g_telemetry.fillhead_state = getCurrentState();
    g_telemetry.injector_state = getInjectorState();
    g_telemetry.temp_c = readTemperature();
    g_telemetry.injector_torque = motor0.GetTorque();
    g_telemetry.injector_homed = motor0.HlfbState() == HLFB_HOMED;
    // ... update other fields
    
    // Send complete telemetry - automatically formatted!
    telemetry_send(&g_telemetry);
    
    delay(100);
}
```
